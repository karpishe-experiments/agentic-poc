// Copy to the reconstructed bridge's test/community-device-login.test.ts.
// Synthetic HTTP responses only; no network, browser login or real credentials.
import { test } from "node:test";
import assert from "node:assert/strict";
import { spawnSync } from "node:child_process";
import fs from "node:fs";
import os from "node:os";
import path from "node:path";

for (const [label, payload, expected] of [
  ["upstream access_token", { access_token: "synthetic-access" }, "synthetic-access"],
  ["Community token", { token: "synthetic-community" }, "synthetic-community"],
  ["access_token precedence", { access_token: "synthetic-access", token: "synthetic-community" }, "synthetic-access"],
  ["blank access_token fallback", { access_token: " ", token: "synthetic-community" }, "synthetic-community"],
  ["missing token fails closed", {}, null],
  ["non-string and blank fail closed", { access_token: 42, token: " " }, null],
] as const) {
  test(`device login: ${label}`, () => {
    const state = fs.mkdtempSync(path.join(os.tmpdir(), "dify-community-auth-fixture-"));
    try {
      const code = `
        globalThis.fetch = async (url) => {
          const pathname = new URL(url).pathname;
          if (pathname === '/openapi/v1/oauth/device/code') return new Response(JSON.stringify({device_code:'synthetic-code',expires_in:60,interval:0}));
          if (pathname === '/openapi/v1/oauth/device/token') return new Response(JSON.stringify(${JSON.stringify(payload)}));
          throw new Error('Unexpected HTTP request');
        };
        process.argv = [process.execPath, 'bin/difywf.js', 'auth', 'login', '--base-url', 'https://synthetic.invalid'];
        await import('./bin/difywf.js');
      `;
      const result = spawnSync(process.execPath, ["--input-type=module", "-e", code], {
        encoding: "utf8", timeout: 15000,
        env: { ...process.env, DIFYWF_HOME: state },
      });
      assert.ifError(result.error);
      const response = JSON.parse(result.stdout);
      const file = path.join(state, "hosts.json");
      if (expected === null) {
        assert.notEqual(result.status, 0);
        assert.equal(response.ok, false);
        assert.equal(response.error.code, "AUTH_REQUIRED");
        assert.equal(fs.existsSync(file), false);
      } else {
        assert.equal(result.status, 0, result.stderr);
        assert.equal(response.ok, true);
        assert.equal(response.data.stored, true);
        assert.equal(JSON.parse(fs.readFileSync(file, "utf8")).hosts["https://synthetic.invalid"].openapi_token, expected);
        assert.ok(!result.stdout.includes(expected));
      }
    } finally {
      fs.rmSync(state, { recursive: true, force: true });
    }
  });
}
